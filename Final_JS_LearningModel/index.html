sketch-p1.js
// face and hand tracking for bg — p5.js sketch
// ml5.js and fractals
let video;
let faceMesh, handPose;
let faces = [], hands = [];
let triangles;
let osc1, osc2;
let notes = [68, 70, 71, 73, 75, 76, 78, 80];
let treeAngle = 0.4;
let branchLen = 80;
let faceReady = false;
let handReady = false;
let bgParticles = [];
const NUM_BG = 180;
let palette = [
  [80, 160, 255], [160, 80, 255],
  [40, 220, 220], [220, 60, 220],
];

function setup() {
  createCanvas(640, 480);
  video = createCapture(VIDEO, { flipped: true });
  video.size(640, 480); video.hide();
  faceMesh = ml5.faceMesh({ maxFaces: 1, flipped: true }, faceModelReady);
  handPose = ml5.handPose({ flipped: true }, handModelReady);
  osc1 = new p5.Oscillator("sine");
  osc2 = new p5.Oscillator("sine");
  osc1.start(); osc2.start();
  osc1.amp(0); osc2.amp(0);
  initParticles();
}

function draw() {
  background(10, 100);
  noiseZ += 0.005;
  // ... particle flow field + face mesh triangles + hand tracking ...
  for (let x = 50; x < width; x += 80) {
    push(); translate(x, height); branch(branchLen); pop();
  }
  filter(POSTERIZE, posterizeAmount);
}

function branch(len) {
  strokeWeight(map(len, 2, 120, 1, 10));
  stroke(219, 181, 55);
  line(0, 0, 0, -len);
  translate(0, -len);
  len *= 0.67;
  if (len > 4) {
    push(); rotate(-treeAngle); branch(len); pop();
    push(); rotate(treeAngle);  branch(len); pop();
  }
}

function faceModelReady() {
  faceMesh.detectStart(video, gotFaces);
  triangles = faceMesh.getTriangles();
  faceReady = true;
}

function handModelReady() {
  handPose.detectStart(video, gotHands);
  handReady = true;
}

function mousePressed() { userStartAudio(); }

How it works

sketch-p2.js
// Piano hand control — p5.js WEBGL sketch
// ml5.js handPose + p5.sound
let video, handPose, hands = [];
let handReady = false;
let fft, audioStarted = false;
let keys = [], keySounds = [];
let lastPlayed = -1;
let notes = [
  'C4.wav', 'D4.wav', 'E4.wav', 'F4.wav',
  'G4.wav', 'A4.wav', 'B4.wav', 'C5.wav'
];

function preload() {
  soundFormats('wav');
  for (let i = 0; i < notes.length; i++) {
    keySounds.push(loadSound(notes[i]));
  }
}

function setup() {
  createCanvas(windowWidth, windowHeight, WEBGL);
  fft = new p5.FFT(0.9, 256);
  video = createCapture(VIDEO);
  video.size(640, 480); video.hide();
  handPose = ml5.handPose({ flipped: true }, handModelReady);
  let startX = -220;
  for (let i = 0; i < 8; i++) {
    keys.push({ x: startX + i * 60, active: false });
  }
}

function draw() {
  background("#070814");
  let bass = fft.getEnergy("bass");
  // ambient glow + waveform visualizer ...
  for (let key of keys) {
    push(); translate(key.x, 240, 0);
    key.active
      ? emissiveMaterial(255, 240, 120)
      : ambientMaterial(255);
    stroke(0); box(55, 220, 25);
    pop(); key.active = false;
  }
  if (hands.length > 0) {
    let finger = hands[0].keypoints[8];
    let thumb  = hands[0].keypoints[4];
    let pinch = dist(finger.x, finger.y, thumb.x, thumb.y);
    let keyIndex = constrain(
      floor(map(finger.x, 0, video.width, 0, keys.length)),
      0, keys.length - 1
    );
    keys[keyIndex].active = true;
    if (pinch < 40 && keyIndex !== lastPlayed) {
      keySounds[keyIndex].play();
      lastPlayed = keyIndex;
    }
    if (pinch > 50) lastPlayed = -1;
  }
}

function mousePressed() {
  if (!audioStarted) { userStartAudio(); audioStarted = true; }
}
